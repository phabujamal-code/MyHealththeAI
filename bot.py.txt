from telegram import Update
from telegram.ext import ApplicationBuilder, CommandHandler, MessageHandler, ContextTypes, filters
import os

# === 1. دالة البدء /start ===
async def start(update: Update, context: ContextTypes.DEFAULT_TYPE):
    user = update.effective_user
    await update.message.reply_text(
        f"👋 أهلاً {user.first_name}!\n"
        "أنا MyHealth AI 🤖\n"
        "مساعدك الذكي للصحة والعادات اليومية.\n\n"
        "اكتب /tip للحصول على نصيحة صحية اليوم!"
    )

# === 2. دالة النصائح اليومية ===
async def tip(update: Update, context: ContextTypes.DEFAULT_TYPE):
    tips = [
        "🥦 تناول الخضروات يوميًا — فهي غنية بالعناصر المضادة للأكسدة.",
        "💧 اشرب 8 أكواب من الماء على الأقل كل يوم!",
        "😴 حاول النوم من 7 إلى 8 ساعات للحفاظ على نشاطك الذهني.",
        "🚶‍♂️ مارس رياضة خفيفة مثل المشي 30 دقيقة يوميًا.",
        "🧘 خذ وقتًا للاسترخاء والتأمل — صحتك النفسية مهمة أيضًا!"
    ]
    import random
    tip_of_the_day = random.choice(tips)
    await update.message.reply_text(f"✨ نصيحة اليوم:\n{tip_of_the_day}")

# === 3. دالة استجابة للرسائل العامة ===
async def echo(update: Update, context: ContextTypes.DEFAULT_TYPE):
    text = update.message.text.lower()
    if "شكرا" in text or "thank" in text:
        await update.message.reply_text("على الرحب 🤍! تذكر أن الصحة تبدأ بخطوة صغيرة.")
    else:
        await update.message.reply_text("اكتب /tip للحصول على نصيحة جديدة 🌿")

# === 4. تشغيل البوت ===
TOKEN = os.getenv("TELEGRAM_BOT_TOKEN")

if not TOKEN:
    print("⚠️ لم يتم العثور على التوكن! تأكد من إضافته في Render → Environment Variables")
else:
    app = ApplicationBuilder().token(TOKEN).build()

    app.add_handler(CommandHandler("start", start))
    app.add_handler(CommandHandler("tip", tip))
    app.add_handler(MessageHandler(filters.TEXT & ~filters.COMMAND, echo))

    print("🚀 MyHealth AI Bot is running...")
    app.run_polling()
