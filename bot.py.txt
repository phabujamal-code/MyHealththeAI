import os
from telegram import Update
from telegram.ext import (
    ApplicationBuilder,
    CommandHandler,
    MessageHandler,
    ContextTypes,
    filters
)

# -------------------------------
# 1️⃣ استدعاء التوكن من إعدادات Render
# -------------------------------
TOKEN = os.getenv("TELEGRAM_BOT_TOKEN")

# -------------------------------
# 2️⃣ دالة الترحيب عند بدء البوت
# -------------------------------
async def start(update: Update, context: ContextTypes.DEFAULT_TYPE):
    await update.message.reply_text(
        "👋 أهلاً بك في بوت GrowMind AI!\n\n"
        "هذا البوت يقدم لك نصيحة تحفيزية يومية لتحسين حياتك ورفع إنتاجيتك.\n"
        "اكتب /tip للحصول على أول نصيحة اليوم 💡"
    )

# -------------------------------
# 3️⃣ دالة النصائح اليومية
# -------------------------------
async def tip(update: Update, context: ContextTypes.DEFAULT_TYPE):
    tips = [
        "✨ ركّز على هدف واحد فقط اليوم وستحقق أكثر مما تتخيل.",
        "🧘‍♂️ لا تبدأ يومك بالهاتف، خذ 10 دقائق للتنفس والتخطيط.",
        "💪 المثابرة أهم من الذكاء. افعل القليل كل يوم.",
        "📖 اقرأ صفحة واحدة يومياً، خلال سنة ستنهي عشرات الكتب.",
        "🚀 كل نجاح يبدأ بخطوة صغيرة، لا تنتظر المثالية.",
    ]
    import random
    message = random.choice(tips)
    await update.message.reply_text(message)

# -------------------------------
# 4️⃣ دالة الردود العامة (لكل رسالة عادية)
# -------------------------------
async def reply(update: Update, context: ContextTypes.DEFAULT_TYPE):
    user_message = update.message.text.lower()
    if "شكراً" in user_message or "thank" in user_message:
        await update.message.reply_text("العفو 🙏 استمر بالتقدم!")
    else:
        await update.message.reply_text("اكتب /tip للحصول على نصيحة جديدة 🌟")

# -------------------------------
# 5️⃣ تشغيل البوت
# -------------------------------
def main():
    app = ApplicationBuilder().token(TOKEN).build()

    app.add_handler(CommandHandler("start", start))
    app.add_handler(CommandHandler("tip", tip))
    app.add_handler(MessageHandler(filters.TEXT & ~filters.COMMAND, reply))

    print("🚀 Bot is running...")
    app.run_polling()

if __name__ == "__main__":
    main()
